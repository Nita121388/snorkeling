[31m+     }[39m

[31m+                 if (isNoControllerFoundError(error)) {[39m
[31m+                     return; // no live controller — frontend repaint already ran[39m
[31m+                 }[39m
[31m+                 console.warn("redrawTUI rpc error", error);[39m
